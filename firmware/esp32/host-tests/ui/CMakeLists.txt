cmake_minimum_required(VERSION 3.16)
project(pixora_ui_smoke C CXX)
set(CMAKE_C_STANDARD 11)
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g -O1)
add_link_options(-fsanitize=address,undefined)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/include/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
set(main "${CMAKE_CURRENT_SOURCE_DIR}/../../main")
set(json "$ENV{IDF_PATH}/components/json/cJSON")
add_executable(ui-smoke test_ui.c "${main}/screen_ui.c" "${main}/screen_i18n.c" "${main}/pixora_cjk_18.c" "${main}/quota.c" "${json}/cJSON.c")
target_include_directories(ui-smoke PRIVATE include "${main}" "${json}")
target_link_libraries(ui-smoke PRIVATE lvgl m)
